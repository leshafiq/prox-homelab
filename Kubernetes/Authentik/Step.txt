1.  helm repo add authentik https://charts.goauthentik.io
    helm repo update
2.  create values.yaml
3.  key secret key authentik-creds
    kubectl create secret generic authentik-creds   
    --from-literal=password='super-password'   
    --from-literal=postgres-password='super-password'   
    --from-literal=replication-password='super-password'   
    --from-literal=secret_key='super-password' 
    -n authentik
4.  helm install authentik authentik/authentik --namespace authentik --create-namespace -f values.yaml
5.  If postgres password is incorrect in logs (FATAL)
    # List the PVCs to find the postgres one
    kubectl get pvc -n authentik
    # Delete the data-postgresql-0 (or similar named) claim
    kubectl delete pvc data-authentik-postgresql-0 -n authentik
6.  install back
7.  Outpost yaml
    authentik_host: https://authentik.xxx.com/
    kubernetes_replicas: 1
    kubernetes_namespace: authentik
    object_naming_template: ak-outpost-%(name)s
    kubernetes_service_type: ClusterIP
    kubernetes_disabled_components:
    - ingress
