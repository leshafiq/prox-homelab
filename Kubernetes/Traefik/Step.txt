1.  Ensure that traefik is not running via K3S (kube-system)
    kubectl get ingressclass
    kubectl delete ingressclass traefik

2.  Create values.yaml file

3.  Run the installer
    helm repo add traefik https://traefik.github.io/charts
    helm repo update
    helm install traefik traefik/traefik \
    --namespace traefik \
    --create-namespace \
    --values values.yaml \
    --skip-crds \
    --set ingressClass.enabled=true

4.  Verify success
    kubectl get ingressclass
    kubectl get pods -n traefik -l app.kubernetes.io/name=traefik
    kubectl get svc -n traefik
    kubectl get ingressroute -n traefik

5.  Once cert manager deployed, create certificate.yaml and then apply and upgrade
    kubectl apply -f certificate.yaml
    helm upgrade traefik traefik/traefik --namespace traefik -f values.yaml

6.  check the certificate status
    kubectl get certificate -n traefik
