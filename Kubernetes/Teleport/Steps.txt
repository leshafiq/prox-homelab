1.  helm repo add teleport https://charts.releases.teleport.dev
    helm repo update
2.  create values.yaml
3.  helm install teleport-cluster teleport/teleport-cluster   -n teleport-cluster   -f values.yaml   --set crds.install=true
4.  Check crds
    kubectl get crds | grep resources.teleport.dev
5.  Create member.yaml
    kubectl exec -i deployment/teleport-cluster-auth -n teleport-cluster -- tctl create -f - < member.yaml
6.  Add user
    kubectl exec -ti deployment/teleport-cluster-auth -n teleport-cluster -- tctl users add shafiq --roles=member,access,editor
7.  create agent-values.yaml
8.  helm upgrade teleport-agent teleport/teleport-kube-agent \
      --namespace teleport-cluster \
      -f agent-values.yaml
9.  create teleport-auth-token secret
    kubectl exec -it teleport-cluster-auth-xxxx-xxxx -n teleport-cluster -- tctl tokens add --type=app --ttl=8760h
    kubectl create secret generic teleport-auth-token --from-literal=auth-token=TOKEN-HERE -n teleport-cluster
10.  kubectl patch secret teleport-auth-token \
  -n teleport-cluster \
  --type merge \
  -p '{
    "metadata": {
      "labels": {
        "app.kubernetes.io/managed-by": "Helm"
      },
      "annotations": {
        "meta.helm.sh/release-name": "teleport-agent",
        "meta.helm.sh/release-namespace": "teleport-cluster"
      }
    }
  }'
