1.  mkdir -p /var/lib/rancher/k3s/server/manifests/
    sudo su
2.  curl https://kube-vip.io/manifests/rbac.yaml > /var/lib/rancher/k3s/server/manifests/kube-vip-rbac.yaml
3.  kubectl apply -f https://kube-vip.io/manifests/rbac.yaml
4.  export VIP=192.168.0.40
5.  export INTERFACE=eth0
6.  KVVERSION=$(curl -sL https://api.github.com/repos/kube-vip/kube-vip/releases | jq -r ".[0].name")
7.  alias kube-vip="ctr image pull ghcr.io/kube-vip/kube-vip:$KVVERSION; ctr run --rm --net-host ghcr.io/kube-vip/kube-vip:$KVVERSION vip /kube-vip"
8.  kube-vip manifest daemonset \
    --interface $INTERFACE \
    --address $VIP \
    --inCluster \
    --taint \
    --controlplane \
    --services \
    --arp \
    --leaderElection
9.  nano /var/lib/rancher/k3s/server/manifests/kube-vip-manifest.yaml
10. paste the content generated, example as below. And then apply it
11. Join other node according to K3S.txt
12. Then install kube-vip-load-balancer
    kubectl apply -f https://raw.githubusercontent.com/kube-vip/kube-vip-cloud-provider/main/manifest/kube-vip-cloud-controller.yaml
13. Set Loadbalance range
    kubectl create configmap -n kube-system kubevip --from-literal range-global=192.168.1.220-192.168.1.230
