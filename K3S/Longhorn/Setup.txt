1.  get the script checker from Longhorn and run it
    curl -sSfL -o longhornctl https://github.com/longhorn/cli/releases/download/v1.10.1/longhornctl-linux-amd64
    chmod +x longhornctl
    ./longhornctl check preflight --kubeconfig ~/.kube/config --namespace default

2.  add another pod for coredns
    kubectl scale deployment coredns -n kube-system --replicas=2

3.  sudo apt install nfs-common -y

4.  sudo apt install cryptsetup -y

5.  sudo apt install dmsetup -y

6.  sudo modprobe nfs

7.  sudo modprobe dm_crypt

8   sudo nano /etc/modules-load.d/k3s-modules.conf
    nfs
    dm_crypt

9.  sudo systemctl enable --now iscsid

10. sudo systemctl status iscsid

11. sudo systemctl stop multipathd multipathd.socket

12. sudo systemctl disable multipathd multipathd.socket

13. helm repo add longhorn https://charts.longhorn.io

14. helm repo update

15. craete values.yaml file

16. helm install longhorn longhorn/longhorn --namespace longhorn-system --create-namespace --values values.yaml
