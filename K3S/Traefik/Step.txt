# =====[ 3 ENSURE TRAEFIK HAS A LOADBALANCER IP ]=============================================
kubectl -n kube-system get svc traefik                               # view current type/IP
kubectl -n kube-system patch svc traefik -p '{"spec":{"type":"LoadBalancer"}}'   # set to LB (idempotent)
kubectl -n kube-system get svc traefik -o wide                        # verify EXTERNAL-IP assigned by MetalLB

# (Optional) Save the Traefik EXTERNAL-IP to a variable for convenience.
export TRAEFIK_IP="$(kubectl -n kube-system get svc traefik -o jsonpath='{.status.loadBalancer.ingress[0].ip}')"
